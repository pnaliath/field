cmake_minimum_required(VERSION 3.25)
project(Field VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(FieldCore STATIC src/FieldCore.cpp)
target_include_directories(FieldCore PUBLIC include)
target_link_libraries(FieldCore PUBLIC Threads::Threads)
add_executable(FieldCoreTests tests/core_tests.cpp)
target_link_libraries(FieldCoreTests PRIVATE FieldCore)
enable_testing()
add_test(NAME FieldCoreTests COMMAND FieldCoreTests)
if(NOT WIN32)
  return()
endif()
add_library(FieldView STATIC src/FieldView.cpp)
target_link_libraries(FieldView PUBLIC FieldCore gdiplus comdlg32 user32 gdi32)
target_compile_definitions(FieldView PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
add_executable(FieldRenderCheck tests/render_check.cpp)
target_link_libraries(FieldRenderCheck PRIVATE FieldView)
include(FetchContent)
FetchContent_Declare(flsdk GIT_REPOSITORY https://github.com/KennyTheBoss05/ModulatorX.git GIT_TAG 000ce2d64c46b302449d1462c7e3a34ed3a15415)
FetchContent_GetProperties(flsdk)
if(NOT flsdk_POPULATED)
  FetchContent_Populate(flsdk)
endif()
set(FLSDK "${flsdk_SOURCE_DIR}/SDK")
add_library(FieldNative SHARED src/FieldNative.cpp "${FLSDK}/fp_plugclass.cpp" "${FLSDK}/fp_cplug.cpp" "${FLSDK}/fp_extra.cpp" "${FLSDK}/fp_pathmanager.cpp")
target_include_directories(FieldNative PRIVATE "${FLSDK}")
target_compile_definitions(FieldNative PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(FieldNative PRIVATE FieldView)
set_target_properties(FieldNative PROPERTIES PREFIX "" OUTPUT_NAME "Field 1 Beta_x64")

set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
FetchContent_Declare(vst3sdk GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git GIT_TAG 3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96)
FetchContent_MakeAvailable(vst3sdk)
smtg_add_vst3plugin(FieldVST3 src/FieldVST3.cpp "${vst3sdk_SOURCE_DIR}/public.sdk/source/vst/vstsinglecomponenteffect.cpp")
target_include_directories(FieldVST3 PRIVATE "${vst3sdk_SOURCE_DIR}")
target_link_libraries(FieldVST3 PRIVATE sdk FieldView comctl32)
smtg_target_configure_version_file(FieldVST3)
if(MSVC)
  foreach(target FieldCore FieldView FieldNative FieldVST3 FieldCoreTests)
    target_compile_options(${target} PRIVATE /W4 /EHsc /permissive-)
  endforeach()
endif()
